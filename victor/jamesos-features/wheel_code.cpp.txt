
  // JamesOS: the first move comes after a short turn (like before); if you keep turning,
  // the cursor keeps moving every 0.35 s, so long lists don't need stop-and-go.
  const auto kJamesWheelRepeat = std::chrono::milliseconds(350);
  bool JamesWheelMove(u32 count)
  {
    static std::chrono::steady_clock::time_point last;
    const auto now = std::chrono::steady_clock::now();
    if (count == kMenuCursorMoveCountThresh) { last = now; return true; }
    if (count > kMenuCursorMoveCountThresh && now - last >= kJamesWheelRepeat) { last = now; return true; }
    return false;
  }
